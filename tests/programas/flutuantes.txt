flut pesos[3];
flut total;

flut media(flut v[], int n) {
   flut s;
   int i;
   s = 0.0;
   i = 0;
   enquanto (i < n) execute { s = s + v[i]; i = i + 1; }
   retorne s / n;
}

flut maior(flut a, flut b) {
   retorne a > b ? a : b;
}

programa {
   flut locais[2];
   pesos[0] = 1.5; pesos[1] = 2.5; pesos[2] = 5.0;
   locais[0] = -1.0; locais[1] = 0.5;
   total = pesos[0] + pesos[1] + pesos[2];
   escreva total; novalinha;
   escreva media(pesos, 3); novalinha;
   escreva media(locais, 2); novalinha;
   escreva maior(1.25, -3.0); novalinha;
   escreva 1 + 0.5; escreva " "; escreva 0.5 * 4; escreva " "; escreva 7 / 2; novalinha;
   escreva 1.5 == 1.5; escreva 1.5 != 1.5; escreva 1.5 < 2.0; escreva 2.0 <= 2.0; escreva -0.5 > 0.0; escreva 3.0 >= 3.5;
   novalinha;
   escreva -total; novalinha;
}
