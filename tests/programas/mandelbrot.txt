car simbolos[10];

vazio inicia_simbolos() {
   /* Símbolos para diferentes níveis de divergência */
   simbolos[0] = ' ';
   simbolos[1] = '.';
   simbolos[2] = ',';
   simbolos[3] = ':';
   simbolos[4] = '-';
   simbolos[5] = '=';
   simbolos[6] = '+';
   simbolos[7] = '*';
   simbolos[8] = '&';
   simbolos[9] = '#';
}

car get_simbolo(int indice) {
   retorne simbolos[indice];
}

programa {
   flut real, imaginario, x, y, xtemp;
   flut realCentro, imaginarioCentro, escalaZoom;
   int iteracao, maxIteracoes;
   int largura, altura;
   int i, j;

   inicia_simbolos();

   /* Configurações de Zoom */
   realCentro = -0.45;
   imaginarioCentro = 0.0;
   escalaZoom = 0.5;

   /* Configurações da tela */
   largura = 75;
   altura = 30;
   maxIteracoes = 80;

   escreva "Conjunto de Mandelbrot:";
   novalinha;

   para (i = 0; i < altura; i = i + 1) execute {

      para (j = 0; j < largura; j = j + 1) execute {
         real = (j - largura / 2.0) * (4.0 / largura) * escalaZoom + realCentro;
         imaginario = (i - altura / 2.0) * (4.0 / altura) * escalaZoom + imaginarioCentro;

         x = 0.0;
         y = 0.0;

         /* Iteração do conjunto de Mandelbrot */
         para (iteracao = 0; x * x + y * y <= 4.0 e iteracao < maxIteracoes; iteracao = iteracao + 1) execute {
            xtemp = x * x - y * y + real;
            y = 2.0 * x * y + imaginario;
            x = xtemp;
         }

         /* Pontos que não divergiram */
         se (iteracao >= maxIteracoes) entao {
            escreva '@';
         } senao {
            int indice;
            /* Escala não linear de divergência */
            indice = piso(9.0 * raiz(1.0 * iteracao / maxIteracoes));

            escreva get_simbolo(indice);
         }
      }

      novalinha;
   }
}
