int retornoEmBlocos(int valor) {
    int vetor[10];

    vetor[0] = -1;

    {
        int grande;
        grande = 100;

        se (valor > 0) entao {
            vetor[0] = grande;
            retorne vetor[0];
        }
    }
    {
        int  v[10], pequeno;
        pequeno = 10;
        v[0] = pequeno;

        se (valor < 0) entao {
            vetor[0] = v[0];
            retorne vetor[0];
        }
    }

    retorne vetor[0];
}

vazio escreveVetor(int v[], int tamanho) {
    int i;
    i = 0;
    enquanto (i < tamanho) execute {
        escreva v[i];
        escreva " ";
        i = i + 1;
    }
    novalinha;
}

vazio blocoIrmao() {
    {
        int v[10], temp;

        temp = 0;
        enquanto (temp < 10) execute {
            v[temp] = 10 - temp;
            temp = temp + 1;
        }

        escreveVetor(v, 10);
    }
    {
        car b;
        int a;

        b = 'b';
        a = 10;
        escreva a;
        escreva b;
        novalinha;
    }
}

programa {
    escreva 1000 + retornoEmBlocos(10);
    novalinha;
    escreva retornoEmBlocos(0);
    novalinha;
    {
        escreva retornoEmBlocos(10 > 7 ? -1 : 20);
        novalinha;
    }
    {
        blocoIrmao();
    }
}
